match /calls/{callId} {
  function isParticipant() {
    return request.auth != null &&
      (request.auth.uid == resource.data.callerId || request.auth.uid == resource.data.calleeId);
  }
  function isFriendCall() {
    return request.auth != null &&
      request.resource.data.callerId == request.auth.uid &&
      request.resource.data.calleeId is string &&
      request.resource.data.calleeId != request.auth.uid &&
      request.resource.data.friendshipId is string &&
      exists(/databases/$(database)/documents/friendships/$(request.resource.data.friendshipId)) &&
      get(/databases/$(database)/documents/friendships/$(request.resource.data.friendshipId)).data.memberUids.hasAll([request.auth.uid, request.resource.data.calleeId]);
  }
  allow create: if isFriendCall();
  allow read: if isParticipant();
  allow update: if isParticipant() &&
    request.resource.data.callerId == resource.data.callerId &&
    request.resource.data.calleeId == resource.data.calleeId;
  allow delete: if false;

  match /callerCandidates/{candidateId} {
    allow read, write: if request.auth != null &&
      (request.auth.uid == get(/databases/$(database)/documents/calls/$(callId)).data.callerId ||
       request.auth.uid == get(/databases/$(database)/documents/calls/$(callId)).data.calleeId);
  }
  match /calleeCandidates/{candidateId} {
    allow read, write: if request.auth != null &&
      (request.auth.uid == get(/databases/$(database)/documents/calls/$(callId)).data.callerId ||
       request.auth.uid == get(/databases/$(database)/documents/calls/$(callId)).data.calleeId);
  }
}
